\begin{abstract}
Traditional automobile insurance pricing relies on demographic proxies (age, gender, postal code, credit history) that penalize safe drivers and fail to capture actual on-road driving dynamics. We present \textbf{INSURE}, an end-to-end telematics pricing architecture that ingests consented real-world vehicle sensor measurements (OBD-II, smartphone accelerometer, gyroscope, GPS), defensively validates erratic or incomplete data, calculates explainable risk scores across 5 physical driving factors, applies regulatory-compliant discount schedules, and continuously audits model demographic fairness. We provide complete mathematical formulations, real sample trip derivations, portfolio economics proofs, and role-based workflows for customers, underwriters, and compliance officers.
\end{abstract}

\subsection{Step 3: Transparent Dynamic Premium Formulation}
Given approved baseline annual premium $P_{\text{base}} = \text{INR } 2,000$:
\begin{equation}
    \text{Discount Schedule } D(S) = 
    \begin{cases}
    10\% \text{ to } 15\% & \text{if } S \ge 90 \quad \text{(Low Risk Tier)} \\
    5\% \text{ to } 10\% & \text{if } 75 \le S < 90 \quad \text{(Moderate Safe Tier)} \\
    0\% & \text{if } 60 \le S < 75 \quad \text{(Standard Market Baseline)} \\
    \le +20\% \text{ (Capped)} & \text{if } S < 60 \quad \text{(High Risk Surcharge Tier)}
    \end{cases}
\end{equation}

For Safety Score $S = 93 \implies D = 10\%$:
\begin{equation}
    P_{\text{final}} = P_{\text{base}} \times (1 - D) = 2,000 \times (1 - 0.10) = \mathbf{\text{INR } 1,800}
\end{equation}
\begin{equation}
    \text{Annual Driver Savings} = 2,000 - 1,800 = \mathbf{\text{INR } 200 \text{ / year (10\% net discount)}}
\end{equation}

\section{Three-Role Live Demonstration Workflow}

\subsection{Role 1: Policyholder / Customer}
\begin{enumerate}[noitemsep]
    \item \textbf{Privacy \& Consent Control} (\texttt{/consent}): Demonstrates uncoerced opt-in, toggleable GPS resolution, and complete data minimization.
    \item \textbf{IoT Driving Simulator} (\texttt{/simulator}): Executes the real-time 5-stage actuarial pipeline (Ingestion $\to$ Validation $\to$ Scoring $\to$ Claim Separation $\to$ Quote).
    \item \textbf{Explainable Billing \& Dispute Channel} (\texttt{/assessments}): Displays itemized factor deductions and enables 1-click human dispute dispatch for anomalous events.
\end{enumerate}

\subsection{Role 2: Underwriter / Insurance Admin}
\begin{enumerate}[noitemsep]
    \item \textbf{Portfolio Growth \& Economics} (\texttt{/profitability}): Interactive actuarial sandbox evaluating Conservative, Expected, and Optimistic presets with live combined ratio calculations.
    \item \textbf{Pricing Optimizer}: Multi-scenario search algorithm picking the highest-profit scenario that satisfies all regulatory discount ceilings.
    \item \textbf{Dispute Resolution Panel} (\texttt{/review}): Underwriters inspect raw event waveforms, override unfair penalties, and trigger real-time premium adjustments.
\end{enumerate}

\subsection{Role 3: Regulatory Compliance Officer}
\begin{enumerate}[noitemsep]
    \item \textbf{Algorithmic Fairness Audit} (\texttt{/audit}): Evaluates Four-Fifths demographic parity across gender, age, and location proxies to guarantee non-discriminatory pricing.
    \item \textbf{Audit Trail Export}: Instant export of signed CSV and JSON compliance bundles for insurance regulatory inspection.
\end{enumerate}
